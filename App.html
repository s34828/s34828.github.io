// ===== รายชื่อหน้าทั้งหมด (แต่ละหน้าเป็นไฟล์ .html จริง แยกกัน) ===== const PAGES = [ { file: "index.html", label: "คำนำ", icon: `` }, { file: "education.html", label: "ประวัติการศึกษา", icon: `` }, { file: "profile.html", label: "ประวัติส่วนตัว", icon: `` }, { file: "activities.html", label: "กิจกรรม", icon: `` }, { file: "interests.html", label: "ความสนใจศึกษาต่อ", icon: `` }, ]; const SITE_NAME = "พอร์ตฟอลิโอ"; // หาว่าไฟล์ปัจจุบันคือหน้าไหนใน PAGES (รองรับทั้ง "" และ "index.html" ว่าคือหน้าแรก) function currentFile(){ let f = location.pathname.split('/').pop(); if (!f) f = "index.html"; return f; } function currentIndex(){ const f = currentFile(); const idx = PAGES.findIndex(p => p.file === f); return idx >= 0 ? idx : 0; } const current = currentIndex(); // ตั้งชื่อแท็บ/หน้าต่างเบราว์เซอร์ให้ตรงกับหน้าที่เปิดอยู่เสมอ document.title = SITE_NAME + " | " + PAGES[current].label; // ===== สร้างแถบเมนูด้านบน (แต่ละแท็บลิงก์ไปหน้าไฟล์จริง) ===== const tabsEl = document.getElementById('tabs'); if (tabsEl) { PAGES.forEach((p, i) => { const tab = document.createElement('a'); tab.className = 'tab' + (i === current ? ' active' : ''); tab.href = p.file; tab.innerHTML = `
${p.label}`; tabsEl.appendChild(tab); }); } // ===== ตัวนับหน้า + แถบความคืบหน้า ===== const countEl = document.getElementById('count'); if (countEl) countEl.textContent = String(current + 1).padStart(2, '0') + " / " + String(PAGES.length).padStart(2, '0'); const progressFill = document.getElementById('progressFill'); if (progressFill) progressFill.style.width = ((current + 1) / PAGES.length * 100) + '%'; // ===== ปุ่มก่อนหน้า/ถัดไป (ลิงก์ไปไฟล์จริง) ===== const prevBtn = document.getElementById('prevBtn'); const nextBtn = document.getElementById('nextBtn'); if (prevBtn) { if (current > 0) { prevBtn.href = PAGES[current - 1].file; } else { prevBtn.classList.add('disabled'); prevBtn.removeAttribute('href'); } } if (nextBtn) { if (current < PAGES.length - 1) { nextBtn.href = PAGES[current + 1].file; } else { nextBtn.classList.add('disabled'); nextBtn.removeAttribute('href'); } } // ===== ปุ่มลูกศรซ้าย/ขวาบนคีย์บอร์ด สลับหน้าจริง ===== document.addEventListener('keydown', (e) => { if (e.key === 'ArrowRight' && current < PAGES.length - 1) location.href = PAGES[current + 1].file; if (e.key === 'ArrowLeft' && current > 0) location.href = PAGES[current - 1].file; }); // ===== ปัดซ้าย/ขวาบนมือถือ สลับหน้าจริง ===== let touchStartX = 0; document.addEventListener('touchstart', e => touchStartX = e.touches[0].clientX); document.addEventListener('touchend', e => { const dx = e.changedTouches[0].clientX - touchStartX; if (Math.abs(dx) > 50) { if (dx < 0 && current < PAGES.length - 1) location.href = PAGES[current + 1].file; if (dx > 0 && current > 0) location.href = PAGES[current - 1].file; } }); // ===== LIGHTBOX: กดดูรูปกิจกรรม (มีกรอบข้อมูลด้านล่าง) / รูปคณะที่สนใจศึกษาต่อ ===== const lightbox = document.getElementById('lightbox'); if (lightbox) { const lightboxImg = document.getElementById('lightboxImg'); const lightboxClose = document.getElementById('lightboxClose'); const lightboxCaption = document.getElementById('lightboxCaption'); const lightboxDate = document.getElementById('lightboxDate'); const lightboxTitle = document.getElementById('lightboxTitle'); const lightboxDesc = document.getElementById('lightboxDesc'); function openLightbox(src, alt, caption){ lightboxImg.src = src; lightboxImg.alt = alt || 'รูปขยาย'; if (caption) { lightboxDate.textContent = caption.date || ''; lightboxDate.style.display = caption.date ? '' : 'none'; lightboxTitle.textContent = caption.title || ''; lightboxDesc.textContent = caption.desc || ''; lightboxCaption.style.display = 'block'; } else { lightboxCaption.style.display = 'none'; } lightbox.classList.add('open'); } function closeLightbox(){ lightbox.classList.remove('open'); lightboxImg.src = ''; } // รูปในหน้ากิจกรรม: กดแล้วโชว์กรอบข้อมูล (ปี/ชื่อกิจกรรม/รายละเอียด) ด้านล่างรูป document.querySelectorAll('.activity-item').forEach(item => { const img = item.querySelector('.activity-thumb img'); if (!img) return; const dateEl = item.querySelector('.date'); const titleEl = item.querySelector('.title'); const descEl = item.querySelector('.soft'); img.addEventListener('click', () => openLightbox( img.getAttribute('src'), img.getAttribute('alt'), { date: dateEl ? dateEl.textContent.trim() : '', title: titleEl ? titleEl.textContent.trim() : '', desc: descEl ? descEl.textContent.trim() : '' } )); }); // รูปคณะ/มหาวิทยาลัยที่สนใจ: กดดูรูปขยายได้ (ไม่มีกรอบข้อความ) document.querySelectorAll('.uni-photo img').forEach(img => { img.addEventListener('click', () => openLightbox(img.getAttribute('src'), img.getAttribute('alt'))); }); lightbox.addEventListener('click', (e) => { if (e.target === lightbox) closeLightbox(); }); if (lightboxClose) lightboxClose.addEventListener('click', closeLightbox); document.addEventListener('keydown', (e) => { if (e.key === 'Escape') closeLightbox(); }); }
